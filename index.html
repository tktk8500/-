import sys from PyQt5.QtWidgets import QApplication, QMainWindow from PyQt5.QAxContainer import QAxWidget class KiwoomHogaWatcher(QMainWindow): def __init__(self): super().__init__() self.kiwoom = QAxWidget("KHOPENAPI.KHOpenAPICtrl.1") # 키움 이벤트 연결 self.kiwoom.OnEventConnect.connect(self.on_event_connect) self.kiwoom.OnReceiveRealData.connect(self.on_receive_real_data) # 감시할 종목코드 (예: 한국전력 015760, 삼성전자 005930) self.target_code = "015760" # 상태 저장 변수 self.current_price = 0 self.strength = 0.0 self.total_sell_qty = 0 self.total_buy_qty = 0 print("키움 OpenAPI 로그인 시도 중...") self.kiwoom.dynamicCall("CommConnect()") def on_event_connect(self, err_code): if err_code == 0: print("✅ 로그인 성공! 실시간 호가 및 체결 데이터 감시를 시작합니다.") self.register_real_data() else: print(f"❌ 로그인 실패 (에러코드: {err_code})") def register_real_data(self): # FID: 10(현재가), 228(체결강도), 121(매도총잔량), 125(매수총잔량), 41(매도1호가)... fid_list = "10;228;121;125;41;51;61;71;81" self.kiwoom.dynamicCall( "SetRealReg(QString, QString, QString, QString)", "0101", self.target_code, fid_list, "0" ) print(f"[{self.target_code}] 실시간 호가 감시 등록 완료") def on_receive_real_data(self, code, real_type, real_data): if code != self.target_code: return # 1. 주식체결 데이터 수신 if real_type == "주식체결": price_raw = self.kiwoom.dynamicCall("GetCommRealData(QString, int)", code, 10) strength_raw = self.kiwoom.dynamicCall("GetCommRealData(QString, int)", code, 228) if price_raw: self.current_price = abs(int(price_raw.strip())) if strength_raw: self.strength = abs(float(strength_raw.strip())) # 2. 주식호가잔량 데이터 수신 elif real_type == "주식호가잔량": total_sell = self.kiwoom.dynamicCall("GetCommRealData(QString, int)", code, 121) total_buy = self.kiwoom.dynamicCall("GetCommRealData(QString, int)", code, 125) if total_sell: self.total_sell_qty = abs(int(total_sell.strip())) if total_buy: self.total_buy_qty = abs(int(total_buy.strip())) # 실시간 수급 진단 실행 self.evaluate_hoga_power() def evaluate_hoga_power(self): if self.total_buy_qty == 0 or self.current_price == 0: return ratio = round((self.total_sell_qty / self.total_buy_qty) * 100, 1) # 세력 매집/돌파 조건: 매도잔량이 매수잔량의 1.5배 이상 & 체결강도 100% 초과 if self.total_sell_qty > (self.total_buy_qty * 1.5) and self.strength >= 100.0: print(f"\n🔥 [강력 상승/매집 신호 포착]") print(f" 현재가: {self.current_price:,}원 | 체결강도: {self.strength}%") print(f" 매도잔량: {self.total_sell_qty:,}주 vs 매수잔량: {self.total_buy_qty:,}주 (비율: {ratio}%)") print(f" ➔ 세력이 위쪽 매도벽을 시장가로 흡수하며 시세 견인 중!") # 허매수/물량받이 의심: 매수잔량만 두텁고 체결강도 100% 미만 elif self.total_buy_qty > (self.total_sell_qty * 1.5) and self.strength < 90.0: print(f"\n⚠️ [허매수 / 물량받이 주의]") print(f" 현재가: {self.current_price:,}원 | 체결강도: {self.strength}%") print(f" 매도잔량: {self.total_sell_qty:,}주 vs 매수잔량: {self.total_buy_qty:,}주") print(f" ➔ 아래 받침만 대놓고 위에서 던지는 물량 넘기기 패턴 의심") if __name__ == "__main__": app = QApplication(sys.argv) watcher = KiwoomHogaWatcher() sys.exit(app.exec_())
